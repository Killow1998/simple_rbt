cmake_minimum_required(VERSION 3.22)
set(CMAKE_CUDA_COMPILER /usr/local/cuda-12.9/bin/nvcc CACHE FILEPATH "CUDA compiler")
set(CMAKE_CUDA_HOST_COMPILER /usr/bin/g++ CACHE FILEPATH "CUDA host compiler")
project(simple_rbt_native LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)

set(PROJECT_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}")
file(GLOB MUJOCO_SEARCH "${PROJECT_WORKSPACE}/.venv/lib/python*/site-packages/mujoco")
if(NOT MUJOCO_SEARCH)
  message(FATAL_ERROR "MuJoCo SDK missing. Run ./setup.sh to prepare the locked SDK libraries.")
endif()
list(GET MUJOCO_SEARCH 0 MUJOCO_DIR)
file(GLOB MUJOCO_LIB "${MUJOCO_DIR}/libmujoco.so.*")
list(GET MUJOCO_LIB 0 MUJOCO_LIBRARY)
file(GLOB ORT_LIB "${PROJECT_WORKSPACE}/.venv/lib/python*/site-packages/onnxruntime/capi/libonnxruntime.so.*")
if(NOT ORT_LIB)
  message(FATAL_ERROR "ONNX Runtime shared library missing. Run ./setup.sh first.")
endif()
list(GET ORT_LIB 0 ONNXRUNTIME_LIBRARY)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
file(CREATE_LINK "${ONNXRUNTIME_LIBRARY}" "${CMAKE_CURRENT_BINARY_DIR}/lib/libonnxruntime.so.1" SYMBOLIC)
set(CMAKE_BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/lib;${MUJOCO_DIR}")
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_DIR}/include")
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ONNXRUNTIME_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime")

# Model keys are <ROS robot name>-<controller>. Read the registration data rather
# than keeping a second list of chassis in the build/test configuration.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/catalog.json" MODEL_CATALOG)
string(JSON MODEL_COUNT LENGTH "${MODEL_CATALOG}" models)
math(EXPR MODEL_LAST "${MODEL_COUNT} - 1")
foreach(INDEX RANGE 0 ${MODEL_LAST})
  string(JSON MODEL_KEY MEMBER "${MODEL_CATALOG}" models ${INDEX})
  string(REGEX REPLACE "-.*$" "" ROBOT "${MODEL_KEY}")
  list(APPEND REGISTERED_ROBOTS "${ROBOT}")
endforeach()
list(REMOVE_DUPLICATES REGISTERED_ROBOTS)
foreach(ROBOT IN LISTS REGISTERED_ROBOTS)
  add_test(NAME ros_command_${ROBOT} COMMAND ros_transport --command-probe ${ROBOT})
  set_tests_properties(ros_command_${ROBOT} PROPERTIES LABELS "ros;robot_${ROBOT}" TIMEOUT 15 RESOURCE_LOCK ros_domain_89 ENVIRONMENT "ROS_DOMAIN_ID=89;ROS_LOCALHOST_ONLY=1")
endforeach()
