find_package(Threads REQUIRED)
add_library(simple_rbt_ipc codec.cpp socket.cpp client.cpp server.cpp)
target_include_directories(simple_rbt_ipc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include" "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
target_compile_features(simple_rbt_ipc PUBLIC cxx_std_17)
target_compile_options(simple_rbt_ipc PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(simple_rbt_ipc PUBLIC Threads::Threads simple_rbt_robots)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/ipc/session.cpp" AND
   (BUILD_TESTING OR CMAKE_PROJECT_NAME STREQUAL "simple_rbt_native"))
  add_executable(ipc_session_test "${CMAKE_CURRENT_SOURCE_DIR}/../tests/ipc/session.cpp")
  target_link_libraries(ipc_session_test PRIVATE simple_rbt_ipc)
  add_test(NAME ipc_session COMMAND ipc_session_test)
  set_tests_properties(ipc_session PROPERTIES TIMEOUT 20 LABELS "ipc;e2e")
endif()
