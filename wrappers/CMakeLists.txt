cmake_minimum_required(VERSION 3.16)
project(simple_rbt_ros_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SIMPLE_RBT_ROS ROS2 CACHE STRING "ROS1 or ROS2")
if(SIMPLE_RBT_ROS STREQUAL "ROS1")
  set(SIMPLE_RBT_ROS_DISTRO noetic)
  set(SIMPLE_RBT_ROS_VERSION 1)
elseif(SIMPLE_RBT_ROS STREQUAL "ROS2")
  set(SIMPLE_RBT_ROS_DISTRO humble)
  set(SIMPLE_RBT_ROS_VERSION 2)
else()
  message(FATAL_ERROR "simple_rbt supports only ROS1 Noetic or ROS2 Humble adapters")
endif()
if(NOT "$ENV{ROS_DISTRO}" STREQUAL "${SIMPLE_RBT_ROS_DISTRO}" OR
   NOT "$ENV{ROS_VERSION}" STREQUAL "${SIMPLE_RBT_ROS_VERSION}")
  message(FATAL_ERROR "This adapter requires ROS${SIMPLE_RBT_ROS_VERSION} ${SIMPLE_RBT_ROS_DISTRO}. Source its setup.bash or build deployment/ros${SIMPLE_RBT_ROS_VERSION}/Dockerfile; the host does not need ROS installed.")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_library(simple_rbt_robots "${ROOT}/src/robots.cpp")
target_include_directories(simple_rbt_robots PUBLIC "${ROOT}/include")
add_subdirectory("${ROOT}/ipc" "${CMAKE_CURRENT_BINARY_DIR}/ipc")
if(SIMPLE_RBT_ROS STREQUAL "ROS1")
  add_library(simple_rbt_ros ros1/ros_bridge.cpp common/names.cpp)
  target_compile_definitions(simple_rbt_ros PRIVATE SIMPLE_RBT_ROS1)
  set(SIMPLE_RBT_ROS1_PREFIX "" CACHE PATH "Optional exported SDK for development validation")
  set(SIMPLE_RBT_ROS1_MESSAGES_PREFIX "${ROOT}/build/interfaces/ros1/install" CACHE PATH "Generated message headers")
  if(SIMPLE_RBT_ROS1_PREFIX)
    file(READ "${SIMPLE_RBT_ROS1_PREFIX}/share/roscpp/package.xml" ROSCPP_MANIFEST)
    string(REGEX MATCH "<version>([^<]+)</version>" ROSCPP_VERSION_MATCH "${ROSCPP_MANIFEST}")
    set(roscpp_VERSION "${CMAKE_MATCH_1}")
    target_include_directories(simple_rbt_ros SYSTEM PRIVATE "${SIMPLE_RBT_ROS1_PREFIX}/include" "${SIMPLE_RBT_ROS1_MESSAGES_PREFIX}/include")
    foreach(LIBRARY roscpp rosconsole rostime roscpp_serialization xmlrpcpp cpp_common)
      find_library(ROS1_${LIBRARY} NAMES ${LIBRARY} PATHS "${SIMPLE_RBT_ROS1_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
      target_link_libraries(simple_rbt_ros PRIVATE "${ROS1_${LIBRARY}}")
    endforeach()
  else()
    find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs nav_msgs rosgraph_msgs std_msgs tf2_msgs livox_ros_driver2)
    target_include_directories(simple_rbt_ros SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
    target_link_libraries(simple_rbt_ros PRIVATE ${catkin_LIBRARIES})
  endif()
  if(roscpp_VERSION VERSION_LESS "1.15.0" OR NOT roscpp_VERSION VERSION_LESS "1.18.0")
    message(FATAL_ERROR "ROS1 Noetic roscpp 1.15-1.17 is required; found ${roscpp_VERSION}")
  endif()
  set(SIMPLE_RBT_ROS_CLIENT_VERSION "${roscpp_VERSION}")
elseif(SIMPLE_RBT_ROS STREQUAL "ROS2")
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  if(rclcpp_VERSION VERSION_LESS "16.0.0" OR NOT rclcpp_VERSION VERSION_LESS "17.0.0")
    message(FATAL_ERROR "ROS2 Humble rclcpp 16.x is required; found ${rclcpp_VERSION}")
  endif()
  set(SIMPLE_RBT_ROS_CLIENT_VERSION "${rclcpp_VERSION}")
  find_package(geometry_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(livox_ros_driver2 REQUIRED)
  add_library(simple_rbt_ros ros2/ros_bridge.cpp common/names.cpp)
  ament_target_dependencies(simple_rbt_ros rclcpp geometry_msgs sensor_msgs nav_msgs rosgraph_msgs std_msgs tf2_ros)
  target_link_libraries(simple_rbt_ros livox_ros_driver2::livox_interfaces2__rosidl_typesupport_cpp)
else()
  message(FATAL_ERROR "Independent wrapper requires SIMPLE_RBT_ROS=ROS1 or ROS2")
endif()
target_include_directories(simple_rbt_ros PUBLIC "${ROOT}/include" "${ROOT}/wrappers/include")
target_compile_options(simple_rbt_ros PRIVATE -Wall -Wextra -Wpedantic)
if(SIMPLE_RBT_ROS STREQUAL "ROS2")
  target_link_libraries(simple_rbt_ros simple_rbt_robots)
else()
  target_link_libraries(simple_rbt_ros PRIVATE simple_rbt_robots)
endif()
add_executable(simple-rbt-ros-session session_main.cpp)
target_link_libraries(simple-rbt-ros-session PRIVATE simple_rbt_ipc simple_rbt_ros)
target_compile_definitions(simple-rbt-ros-session PRIVATE
  SIMPLE_RBT_ROS_DISTRO="${SIMPLE_RBT_ROS_DISTRO}"
  SIMPLE_RBT_ROS_VERSION="${SIMPLE_RBT_ROS_VERSION}"
  SIMPLE_RBT_ROS_CLIENT_VERSION="${SIMPLE_RBT_ROS_CLIENT_VERSION}")
if(SIMPLE_RBT_ROS STREQUAL "ROS1")
  target_compile_definitions(simple-rbt-ros-session PRIVATE SIMPLE_RBT_ROS1)
  if(SIMPLE_RBT_ROS1_PREFIX)
    target_link_options(simple-rbt-ros-session PRIVATE "-Wl,-rpath-link,${SIMPLE_RBT_ROS1_PREFIX}/lib")
  endif()
endif()
install(TARGETS simple-rbt-ros-session RUNTIME DESTINATION bin)
option(SIMPLE_RBT_IPC_TEST_FIXTURE "Build independent ROS adapter E2E fixture" OFF)
if(SIMPLE_RBT_IPC_TEST_FIXTURE)
  add_executable(ros_ipc_fixture "${ROOT}/tests/ipc/ros_fixture.cpp")
  target_link_libraries(ros_ipc_fixture PRIVATE simple_rbt_ipc)
endif()
