cmake_minimum_required(VERSION 3.16)
project(sprobo_tutorial_observer LANGUAGES CXX)
add_executable(simple-rbt-tutorial-observer main.cpp)
target_compile_features(simple-rbt-tutorial-observer PRIVATE cxx_std_17)
target_include_directories(simple-rbt-tutorial-observer PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if("$ENV{ROS_VERSION}" STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs sensor_msgs rosgraph_msgs)
  target_compile_definitions(simple-rbt-tutorial-observer PRIVATE ROS1)
  target_include_directories(simple-rbt-tutorial-observer PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(simple-rbt-tutorial-observer ${catkin_LIBRARIES})
else()
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  ament_target_dependencies(simple-rbt-tutorial-observer rclcpp nav_msgs sensor_msgs rosgraph_msgs)
endif()
install(TARGETS simple-rbt-tutorial-observer RUNTIME DESTINATION bin)
